CREATE OR REPLACE FUNCTION public.lista_poa_nacional(IN gestion integer)
  RETURNS TABLE(
    dep_id numeric, dep_departamento character varying, dep_cod character varying, 
    dist_id integer, dist_distrital character varying, dist_cod character varying, 
    abrev character varying, da character varying, ue character varying, 
    prog character varying, proy character varying, act character varying, 
    proy_id numeric, proy_estado integer, actividad character varying, 
    tipo character varying, proy_sisin character varying, proy_nombre character varying, 
    aper_id numeric, aper_proy_estado integer, pfec_id numeric, 
    pfec_estado integer, aper_gestion integer, tp_id numeric, por_id integer
  ) AS
$BODY$
BEGIN
    RETURN QUERY
    SELECT 
        d.dep_id, d.dep_departamento, d.dep_cod,
        ds.dist_id, ds.dist_distrital, ds.dist_cod, ds.abrev, ds.da, ds.ue,
        apg.aper_programa AS prog, 
        apg.aper_proyecto AS proy, 
        apg.aper_actividad AS act,
        p.proy_id, p.proy_estado, 
        ua.act_descripcion AS actividad, 
        te.tipo, p.proy_sisin, p.proy_nombre,
        apg.aper_id, apg.aper_proy_estado, 
        pfe.pfec_id, pfe.pfec_estado, apg.aper_gestion, 
        p.tp_id, p.por_id
    FROM _proyectofaseetapacomponente pfe
    INNER JOIN aperturaprogramatica apg ON apg.aper_id = pfe.aper_id
    INNER JOIN _proyectos p ON p.proy_id = pfe.proy_id
    INNER JOIN _departamentos d ON d.dep_id = p.dep_id
    INNER JOIN _distritales ds ON ds.dist_id = p.dist_id
    INNER JOIN unidad_actividad ua ON ua.act_id = p.act_id
    INNER JOIN v_tp_establecimiento te ON te.te_id = ua.te_id
    WHERE p.estado != 3 
      AND apg.aper_gestion = gestion 
      AND apg.aper_estado != 3 
      AND pfe.pfec_estado = 1
      -- REVISIÓN DE RENDIMIENTO: Si tus tablas de fases guardan histórico, añade este filtro:
      -- AND pfe.gestion = gestion 
    ORDER BY 
        d.dep_id ASC,
        ds.dist_id ASC,
        apg.aper_programa ASC,
        apg.aper_proyecto ASC,
        apg.aper_actividad ASC,
        p.tp_id ASC;
END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100
  ROWS 1000;

CREATE OR REPLACE VIEW vista_seguimiento_partidas AS
SELECT 
    COALESCE(asig.g_id, prog.g_id) AS g_id, -- Columna de Gestión agregada
    COALESCE(asig.aper_id, prog.aper_id) AS aper_id,
    COALESCE(asig.dep_id, prog.dep_id) AS dep_id,
    COALESCE(asig.dist_id, prog.dist_id) AS dist_id,
    COALESCE(asig.sp_id, 0) AS sp_id,
    COALESCE(asig.par_id, prog.par_id) AS par_id,
    COALESCE(asig.partida, prog.codigo::VARCHAR) AS codigo_partida,
    COALESCE(asig.par_nombre, prog.partida_nombre, 'SIN ASIGNAR') AS partida,
    
    COALESCE(asig.ppto_asignado1, 0) AS ppto_asignado,
    COALESCE(prog.ppto_programado1, 0) AS ppto_programado,
    (COALESCE(asig.ppto_asignado1, 0) - COALESCE(prog.ppto_programado1, 0)) AS saldo_poa,
    
    CASE 
        WHEN (COALESCE(asig.ppto_asignado1, 0) - COALESCE(prog.ppto_programado1, 0)) < 0 THEN 3
        WHEN (COALESCE(asig.ppto_asignado1, 0) - COALESCE(prog.ppto_programado1, 0)) > 0 THEN 1
        ELSE 0 
    END AS estado_poa,
    
    COALESCE(asig.ppto_asignado_revertido2, 0) AS ppto_asignado_revertido,
    COALESCE(prog.ppto_programado_revertido2, 0) AS ppto_programado_revertido,
    (COALESCE(asig.ppto_asignado_revertido2, 0) - COALESCE(prog.ppto_programado_revertido2, 0)) AS saldo_revertido,
    
    CASE 
        WHEN (COALESCE(asig.ppto_asignado_revertido2, 0) - COALESCE(prog.ppto_programado_revertido2, 0)) < 0 THEN 3
        WHEN (COALESCE(asig.ppto_asignado_revertido2, 0) - COALESCE(prog.ppto_programado_revertido2, 0)) > 0 THEN 1
        ELSE 0 
    END AS estado_revertido

FROM (
    SELECT 
        pg.sp_id,pg.aper_id, pg.g_id, pg.par_id, pg.partida, par.par_nombre, 
        pg.importe AS ppto_asignado1, pg.ppto_saldo_ncert AS ppto_asignado_revertido2,
        proy.dep_id, proy.dist_id
    FROM ptto_partidas_sigep pg
    INNER JOIN partidas par ON par.par_id = pg.par_id
    INNER JOIN aperturaproyectos ap ON ap.aper_id = pg.aper_id 
    INNER JOIN _proyectos proy ON proy.proy_id = ap.proy_id
    WHERE pg.estado != 3
) asig
FULL OUTER JOIN (
    SELECT 
        i.aper_id, apg.aper_gestion AS g_id, i.par_id, par.par_codigo AS codigo, par.par_nombre AS partida_nombre, 
        proy.dep_id, proy.dist_id,
        SUM(CASE WHEN i.ins_tipo_modificacion = 0 THEN i.ins_costo_total ELSE 0 END) AS ppto_programado1,
        SUM(CASE WHEN i.ins_tipo_modificacion = 1 THEN i.ins_costo_total ELSE 0 END) AS ppto_programado_revertido2
    FROM insumos i
    INNER JOIN partidas par ON par.par_id = i.par_id
    INNER JOIN aperturaproyectos ap ON ap.aper_id = i.aper_id
    INNER JOIN aperturaprogramatica apg ON apg.aper_id = ap.aper_id
    INNER JOIN _proyectos proy ON proy.proy_id = ap.proy_id
    WHERE i.aper_id != 0 AND i.ins_tipo_modificacion IN (0, 1)
    GROUP BY i.aper_id, apg.aper_gestion, i.par_id, par.par_codigo, par.par_nombre, proy.dep_id, proy.dist_id
) prog ON asig.par_id = prog.par_id AND asig.aper_id = prog.aper_id AND asig.g_id = prog.g_id;

    MAX(CASE WHEN ti.mes_id = 6 THEN (CASE WHEN cti.ctins_id IS NOT NULL THEN 1 ELSE 0 END) ELSE 0 END) AS certmes6,
    MAX(CASE WHEN ti.mes_id = 7 THEN (CASE WHEN cti.ctins_id IS NOT NULL THEN 1 ELSE 0 END) ELSE 0 END) AS certmes7,
    MAX(CASE WHEN ti.mes_id = 8 THEN (CASE WHEN cti.ctins_id IS NOT NULL THEN 1 ELSE 0 END) ELSE 0 END) AS certmes8,
    MAX(CASE WHEN ti.mes_id = 9 THEN (CASE WHEN cti.ctins_id IS NOT NULL THEN 1 ELSE 0 END) ELSE 0 END) AS certmes9,

CREATE OR REPLACE FUNCTION public.get_formulario4_consolidado_nacional(IN gest integer)
  RETURNS TABLE(
    -- [1 al 24] Datos de la Apertura / POA (Incrementado con dep_cod y dist_cod)
    dep_id numeric, dep_cod character varying, dep_departamento character varying, 
    dist_id integer, dist_cod character varying, dist_distrital character varying, abrev character varying, 
    da character varying, ue character varying, aper_id numeric, aper_programa character varying, aper_proyecto character varying, 
    aper_actividad character varying, proy_id numeric, proy_sisin character varying, proy_estado integer, tipo character varying, proy_nombre character varying, 
    aper_proy_estado integer, pfec_id numeric, pfec_estado integer, aper_gestion integer, tp_id integer, por_id integer,
    
    -- [25 al 28] Componentes y Subactividad (Recorridos dos espacios)
    com_id numeric, com_componente text, tp_sact integer, tipo_subactividad character varying,
    
    -- [29 al 47] Datos del Producto / Formulario 4
    prod_id numeric, prod_priori integer, prod_producto text, prod_ppto numeric, indi_id numeric, 
    prod_indicador text, prod_linea_base numeric, prod_meta numeric, prod_fuente_verificacion text, 
    prod_unidades character varying, estado integer, prod_mod integer, prod_resultado text, prod_cod integer, 
    uni_resp character varying, prod_observacion text, mt_id integer, or_id integer, sigep_priori integer,
    
    -- [48 al 52] Indicadores y Objetivos
    indi_descripcion character varying, indi_abreviacion character varying,
    or_codigo integer, og_id integer, og_codigo integer, 
    
    -- [53 al 66] Programación Mensual
    m1 numeric, m2 numeric, m3 numeric, m4 numeric, m5 numeric, m6 numeric, 
    m7 numeric, m8 numeric, m9 numeric, m10 numeric, m11 numeric, m12 numeric, 
    g_id integer, total_anual numeric,

    -- [67 al 78] Ejecución Mensual
    ejec_m1 numeric, ejec_m2 numeric, ejec_m3 numeric, ejec_m4 numeric, ejec_m5 numeric, ejec_m6 numeric, 
    ejec_m7 numeric, ejec_m8 numeric, ejec_m9 numeric, ejec_m10 numeric, ejec_m11 numeric, ejec_m12 numeric
  ) AS
$BODY$
BEGIN
    RETURN QUERY
    WITH cte_objetivos AS (
        SELECT DISTINCT 
            ore.or_id, 
            ore.or_codigo, 
            og.og_id, 
            og.og_codigo
        FROM objetivos_regionales ore
        INNER JOIN objetivo_programado_mensual opm ON ore.pog_id = opm.pog_id
        INNER JOIN objetivo_gestion og ON og.og_id = opm.og_id
    )
    SELECT 
        -- [1 al 24] Datos de la Apertura / POA (Casteo explícito de las 24 columnas iniciales)
        poa.dep_id::numeric,
        poa.dep_cod::character varying, 
        poa.dep_departamento::character varying, 
        poa.dist_id::integer, 
        poa.dist_cod::character varying, 
        poa.dist_distrital::character varying, 
        poa.abrev::character varying, 
        poa.da::character varying, 
        poa.ue::character varying, 
        poa.aper_id::numeric, 
        poa.prog::character varying AS aper_programa, 
        poa.proy::character varying AS aper_proyecto, 
        poa.act::character varying AS aper_actividad, 
        poa.proy_id::numeric, 
        poa.proy_sisin::character varying, 
        poa.proy_estado::integer,
        poa.tipo::character varying, 
        poa.proy_nombre::character varying, 
        poa.aper_proy_estado::integer, 
        poa.pfec_id::numeric,
        poa.pfec_estado::integer,
        poa.aper_gestion::integer,
        poa.tp_id::integer, 
        poa.por_id::integer,
        
        -- [25 al 28] Componentes y Subactividad
        c.com_id::numeric, 
        c.com_componente::text, 
        c.tp_sact::integer, 
        sa.tipo_subactividad::character varying,
        
        -- [29 al 47] Datos del Producto / Formulario 4
        p.prod_id::numeric, 
        p.prod_priori::integer, 
        p.prod_producto::text, 
        p.prod_ppto::numeric, 
        p.indi_id::numeric, 
        p.prod_indicador::text, 
        p.prod_linea_base::numeric, 
        p.prod_meta::numeric, 
        p.prod_fuente_verificacion::text, 
        p.prod_unidades::character varying, 
        p.estado::integer, 
        p.prod_mod::integer, 
        p.prod_resultado::text, 
        p.prod_cod::integer, 
        p.uni_resp::character varying, 
        p.prod_observacion::text, 
        p.mt_id::integer, 
        p.or_id::integer, 
        p.sigep_priori::integer,
        
        -- [48 al 52] Indicadores y Objetivos
        i.indi_descripcion::character varying, 
        i.indi_abreviacion::character varying,
        obj.or_codigo::integer, 
        obj.og_id::integer, 
        obj.og_codigo::integer, 
        
        -- [53 al 66] PROGRAMACIÓN MENSUAL (Normalizado a numeric)
        COALESCE(prog.mes1, 0::numeric)::numeric, 
        COALESCE(prog.mes2, 0::numeric)::numeric, 
        COALESCE(prog.mes3, 0::numeric)::numeric, 
        COALESCE(prog.mes4, 0::numeric)::numeric, 
        COALESCE(prog.mes5, 0::numeric)::numeric, 
        COALESCE(prog.mes6, 0::numeric)::numeric, 
        COALESCE(prog.mes7, 0::numeric)::numeric, 
        COALESCE(prog.mes8, 0::numeric)::numeric, 
        COALESCE(prog.mes9, 0::numeric)::numeric, 
        COALESCE(prog.mes10, 0::numeric)::numeric, 
        COALESCE(prog.mes11, 0::numeric)::numeric, 
        COALESCE(prog.mes12, 0::numeric)::numeric, 
        prog.g_id::integer, 
        COALESCE(prog.total_anual, 0::numeric)::numeric,

        -- [67 al 78] EJECUCIÓN MENSUAL
        COALESCE(ejec.e_mes1, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes2, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes3, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes4, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes5, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes6, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes7, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes8, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes9, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes10, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes11, 0::numeric)::numeric, 
        COALESCE(ejec.e_mes12, 0::numeric)::numeric
        
    FROM lista_poa_nacional(gest) poa
    INNER JOIN _componentes c ON c.pfec_id = poa.pfec_id
    INNER JOIN tipo_subactividad sa ON sa.tp_sact = c.tp_sact
    INNER JOIN _productos p ON p.com_id = c.com_id
    INNER JOIN indicador i ON i.indi_id = p.indi_id

    INNER JOIN vista_temporalidad_form4_programado_uresp prog 
        ON prog.prod_id = p.prod_id 
        AND prog.g_id = poa.aper_gestion

    LEFT JOIN vista_temporalidad_form4_ejecutado_uresp ejec 
        ON ejec.prod_id = p.prod_id 
        AND ejec.g_id = poa.aper_gestion

    INNER JOIN cte_objetivos obj ON obj.or_id = p.or_id
    ORDER BY poa.dep_id ASC, poa.dist_id ASC, c.com_id ASC, p.prod_cod ASC;
END;
$BODY$
  LANGUAGE plpgsql VOLATILE
  COST 100
  ROWS 1000;

CREATE OR REPLACE VIEW public.v_temporalidad_diagnostico_equipamiento_quinquenal AS 
 SELECT 
    t.form_equip_id,
    -- Agregación Condicional Numérica Directa por cada gestión del plan estructural
    COALESCE(SUM(CASE WHEN t.g_id = 2026 THEN t.prog_equi END), 0::numeric) AS g_2026,
    COALESCE(SUM(CASE WHEN t.g_id = 2027 THEN t.prog_equi END), 0::numeric) AS g_2027,
    COALESCE(SUM(CASE WHEN t.g_id = 2028 THEN t.prog_equi END), 0::numeric) AS g_2028,
    COALESCE(SUM(CASE WHEN t.g_id = 2029 THEN t.prog_equi END), 0::numeric) AS g_2029,
    COALESCE(SUM(CASE WHEN t.g_id = 2030 THEN t.prog_equi END), 0::numeric) AS g_2030,
    -- Sumatoria analítica acumulada del quinquenio
    COALESCE(SUM(t.prog_equi), 0::numeric) AS total_quinquenal,
    COUNT(*) AS registros_temporales
 FROM public.temporalidad_diagnostico_equipamiento t
 GROUP BY t.form_equip_id;

  OWNER TO postgres;


  //////////////// Lista POa Institucional
  

CREATE OR REPLACE FUNCTION public.fn_lista_poa_nacional(p_gestion INT)
RETURNS TABLE (
    dep_id numeric,
    dep_departamento character varying,
    dep_cod character varying,
    dist_id integer,
    dist_distrital character varying,
    dist_cod character varying,
    abrev character varying,
    da character varying,
    ue character varying,
    aper_programa character varying,
    aper_proyecto character varying,
    aper_actividad character varying,
    proy_id numeric,
    proy_estado integer,
    act_descripcion character varying,
    te_id integer,
    tipo character varying,
    ta_id integer,
    tipo_adm character varying,
    tn_id integer,
    nivel character varying,
    proy_sisin character varying,
    proy_nombre character varying,
    aper_id numeric,
    aper_proy_estado integer,
    pfec_id numeric,
    pfec_descripcion character varying,
    fas_id numeric,
    eta_id numeric,
    pfec_estado integer,
    aper_gestion integer,
    tp_id numeric,
    por_id integer,
    fase_nombre text,
    etapa_nombre TEXT,
    ppto_asignado NUMERIC,
    ppto_poa NUMERIC,
    ppto_saldo NUMERIC,
    tipo_gasto_nombre TEXT,
    estado_poa TEXT
) AS $$
BEGIN -- 🛠️ CORRECCIÓN 1: Bloque de apertura obligatorio añadido
    RETURN QUERY -- 🛠️ CORRECCIÓN 2: Directiva de retorno obligatoria añadida
    WITH 
    -- 1. Consolidado del Presupuesto Asignado desde el SIGEP
    cte_sigep AS (
        SELECT 
            ps.aper_id, -- Especificado alias interno por higiene
            SUM(ps.importe) AS ppto_asignado
        FROM public.ptto_partidas_sigep ps
        GROUP BY ps.aper_id
    ),

    -- 2. Consolidado del Presupuesto Programado POA (Insumos Modificación 0)
    cte_insumos AS (
        SELECT 
            i.aper_id, -- Especificado alias interno por higiene
            SUM(i.ins_costo_total) AS ppto_poa
        FROM public.insumos i
        WHERE i.ins_tipo_modificacion = 0
        GROUP BY i.aper_id
    )

    -- 3. Consulta Principal con Mapeo de Fases y Etapas
    SELECT 
        d.dep_id::numeric, 
        d.dep_departamento::character varying, 
        d.dep_cod::character varying,
        ds.dist_id::integer, 
        ds.dist_distrital::character varying, 
        ds.dist_cod::character varying, 
        ds.abrev::character varying, 
        ds.da::character varying, 
        ds.ue::character varying,
        apg.aper_programa::character varying, 
        apg.aper_proyecto::character varying, 
        apg.aper_actividad::character varying,
        p.proy_id::numeric, 
        p.proy_estado::integer, 
        ua.act_descripcion::character varying, 
    te.te_id::integer,
        te.tipo::character varying,
        te.ta_id::integer,
        te.tipo_adm::character varying,
        te.tn_id::integer,
        te.nivel::character varying, 
        p.proy_sisin::character varying, 
        p.proy_nombre::character varying,
        apg.aper_id::numeric, 
        apg.aper_proy_estado::integer, 
        pfe.pfec_id::numeric,
        pfe.pfec_descripcion::character varying,
        pfe.fas_id::numeric,
        pfe.eta_id::numeric, 
        pfe.pfec_estado::integer, 
        apg.aper_gestion::integer, 
        p.tp_id::numeric, 
        p.por_id::integer,
        
        -- Mapeo de pfe.fas_id (Fase del Requerimiento) con casteo explícito a text
        CASE 
            WHEN pfe.fas_id = 1 THEN 'PREINVERSIÓN'::text
            WHEN pfe.fas_id = 2 THEN 'INVERSIÓN'::text
            ELSE 'NINGUNO'::text
        END AS fase_nombre,

        -- Mapeo de pfe.eta_id (Etapa del Requerimiento) con casteo explícito a text
        CASE 
            WHEN pfe.eta_id = 1 THEN 'ITCP'::text
            WHEN pfe.eta_id = 2 THEN 'TESA'::text
            WHEN pfe.eta_id = 3 THEN 'INVERSION'::text
            WHEN pfe.eta_id = 4 THEN 'OPERACION'::text
            WHEN pfe.eta_id = 5 THEN 'ESTUDIO'::text
            WHEN pfe.eta_id = 6 THEN 'EJECUCION'::text
            WHEN pfe.eta_id = 7 THEN 'FUNCIONAMIENTO'::text
            WHEN pfe.eta_id = 8 THEN 'EDTP'::text
            ELSE 'NINGUNO'::text
        END AS etapa_nombre,

        -- Valores Presupuestarios Consolidados desde las CTEs castrados a NUMERIC
        COALESCE(ppto.ppto_asignado, 0)::numeric AS ppto_asignado, 
        COALESCE(poa_ins.ppto_poa, 0)::numeric AS ppto_poa,
        (COALESCE(ppto.ppto_asignado, 0) - COALESCE(poa_ins.ppto_poa, 0))::numeric AS ppto_saldo,
        
        -- Clasificaciones Organizacionales y de Estado castrados a TEXT
        CASE 
            WHEN p.tp_id = 1 THEN 'INVERSIÓN'::text
            WHEN p.tp_id = 4 THEN 'GASTO CORRIENTE'::text
            ELSE 'OTRO'::text
        END AS tipo_gasto_nombre,
        CASE 
            WHEN apg.aper_proy_estado = 1 THEN 'ANTEPROYECTO'::text
            WHEN apg.aper_proy_estado = 4 THEN 'APROBADO'::text
            ELSE 'OBSERVADO'::text
        END AS estado_poa

    FROM public._proyectofaseetapacomponente pfe
    INNER JOIN public.aperturaprogramatica apg ON apg.aper_id = pfe.aper_id
    INNER JOIN public._proyectos p ON p.proy_id = pfe.proy_id
    INNER JOIN public._departamentos d ON d.dep_id = p.dep_id
    INNER JOIN public._distritales ds ON ds.dist_id = p.dist_id
    INNER JOIN public.unidad_actividad ua ON ua.act_id = p.act_id
    INNER JOIN public.v_tp_establecimiento te ON te.te_id = ua.te_id

    -- 🛠️ CORRECCIÓN 3: Especificación estricta de alias sobre apg.aper_id para romper ambigüedad
    LEFT JOIN cte_sigep ppto ON apg.aper_id = ppto.aper_id
    LEFT JOIN cte_insumos poa_ins ON apg.aper_id = poa_ins.aper_id

    WHERE p.estado != 3 
      AND apg.aper_gestion = p_gestion
      AND apg.aper_estado != 3 
      AND pfe.pfec_estado = 1 

    ORDER BY 
        d.dep_id ASC,
        ds.dist_id ASC,
        apg.aper_programa ASC,
        apg.aper_proyecto ASC,
        apg.aper_actividad ASC,
        p.tp_id ASC;
END;
$$ LANGUAGE plpgsql;
